### 2.1 `bee_tsp/features.py` (new)

```python
# bee_tsp/features.py
import numpy as np
from sklearn.neighbors import KDTree

def compute_features(coords):
    # coords: (n,2) ndarray (assume pre-normalized is OK; if not, center & RMS-scale)
    n = coords.shape[0]
    # bbox & area
    xmin, ymin = coords.min(axis=0); xmax, ymax = coords.max(axis=0)
    A = max((xmax - xmin) * (ymax - ymin), 1e-9)

    # 1-NN scale (exclude self)
    kdt = KDTree(coords)
    d, _ = kdt.query(coords, k=2)
    r1 = d[:,1].mean()

    # density index (~1 uniform; <1 clustered; >1 diffuse)
    delta = 2.0 * r1 * np.sqrt(n / A)

    # PCA anisotropy
    X = coords - coords.mean(axis=0)
    S = np.cov(X.T)
    evals = np.linalg.eigvalsh(S)
    lam_min = max(evals[0], 1e-12)
    anisotropy = float(evals[-1] / lam_min)

    # convex hull fraction
    try:
        from scipy.spatial import ConvexHull
        hull_frac = len(ConvexHull(coords).vertices) / float(n)
    except Exception:
        hull_frac = 0.0

    # cheap cluster score via k-NN modularity proxy
    # build 10-NN graph and measure average neighbor distance vs global mean
    d10, _ = kdt.query(coords, k=min(11, n))
    local_mean = d10[:,1:].mean(axis=1).mean()
    cluster_score = max(0.0, min(1.0, float((r1 - local_mean) / max(r1, 1e-9))))  # crude ∈[0,1]

    return {
        "n": int(n),
        "area": float(A),
        "r1_mean": float(r1),
        "density_idx": float(delta),
        "anisotropy": float(anisotropy),
        "hull_frac": float(hull_frac),
        "cluster_score": float(cluster_score),
    }
```

### 2.2 `bee_tsp/auto_params.py` — add `autoscale_v2(base_cfg, feats)`

```python
# bee_tsp/auto_params.py
def autoscale_v2(base, feats):
    cfg = dict(base)  # shallow copy is OK for our keys
    n = feats["n"]; delta = feats["density_idx"]; a = feats["anisotropy"]
    h = feats["hull_frac"]; C = feats["cluster_score"]

    # size tiers (keep v1 logic, then apply deltas)
    if n <= 6000:    k, zone, ov, agents, bee = 36, None, 0.0, 6, 0.8
    elif n <= 10000: k, zone, ov, agents, bee = 28, 3000, 0.005, 4, 0.6
    elif n <= 30000: k, zone, ov, agents, bee = 26, 3000, 0.005, 4, 0.6
    elif n <= 60000: k, zone, ov, agents, bee = 26, 3000, 0.004, 4, 0.6  # new band (e.g., pla33810)
    else:            k, zone, ov, agents, bee = 24, 3000, 0.003, 3, 0.5

    # sparsity/topology deltas
    if delta < 0.9:                 # clustered
        k += 6; ov += 0.003
        if zone: zone = min(zone, 2500)
        cfg.setdefault("solver", {}).setdefault("routing", {})["cfrs"] = True
    if delta > 1.3:                 # diffuse/outliers
        k -= 4; ov = max(ov - 0.002, 0.0)
    if a > 8.0:                     # corridor
        k += 2
        cfg.setdefault("solver", {}).setdefault("zones", {})["stripe_partitions"] = True
    if h > 0.25:                    # ring/perimeter
        ov += 0.002
        cfg.setdefault("solver", {}).setdefault("traps", {})["bee_balls"] = True
        cfg["solver"].setdefault("candidate", {})["alpha_near_cross"] = 1.5
    if C > 0.35:                    # strong clusters
        cfg.setdefault("solver", {}).setdefault("routing", {})["cfrs"] = True

    # apply back
    sol = cfg.setdefault("solver", {})
    cand = sol.setdefault("candidate", {})
    cand["k"] = max(18, int(k))
    z = sol.setdefault("zones", {})
    z["target_zone_size"] = zone or 999999
    z["overlap_pct"] = float(min(ov, 0.01))
    z["agents_per_zone"] = agents
    sol.setdefault("bees", {})["time_budget_s"] = bee
    # always ensure delaunay + knn on small/mid
    cand.setdefault("use_knn", True)
    if n <= 6000: cand["use_delaunay"] = True
    return cfg
```

## 3) Adaptive overlap + portal nodes (zone boundary quality)

**Where:** in your zoning/partition routine (where overlap gets applied)

**Algorithm:**

* Compute `ring_w = scale * mean(r_k)` with `k≈10` (mean 10-NN distance).
* Set overlap halo as: `halo_pct = min(0.01, ring_w / sqrt(area))`. If user set `overlap_pct`, take `max(user, halo_pct)`.
* **Portal nodes per zone:** score each boundary node by `(external_kNN_degree + EHM_edge_count)` and keep top `p%` (start `p=2`). Force-include these portals in both adjacent zones’ candidate sets.

**Config keys (add):**

```yaml
solver:
  zones:
    adaptive_overlap: { enabled: true, ring_scale: 1.0, portal_pct: 2.0 }
```
