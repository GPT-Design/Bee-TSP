def sample_tour(self, dist, candidate_adj, temperature=0.7, smooth_eps=1.0):
    # weights ~ ((count+smooth_eps)/cost)^(1/temperature); greedy sampling over candidate_adj
    n = self.n
    start = random.randrange(n)
    used = [False]*n; used[start] = True
    tour = [start]
    for _ in range(n-1):
        u = tour[-1]
        cands = [v for v in candidate_adj.get(u, []) if not used[v]]
        if not cands:
            cands = [i for i in range(n) if not used[i]]
        w = []
        for v in cands:
            key = (u,v) if u < v else (v,u)
            cnt = self.counts.get(key, 0)
            score = (cnt + smooth_eps) / (dist(u, v) + 1.0)
            w.append(score ** (1.0 / max(1e-6, temperature)))
        r = random.random() * sum(w); acc = 0.0; pick = cands[-1]
        for v, wi in zip(cands, w):
            acc += wi
            if r <= acc:
                pick = v; break
        used[pick] = True
        tour.append(pick)
    return tour
